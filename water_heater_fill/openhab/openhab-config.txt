// ===== /etc/openhab/rules/items.rules (after the UtilityRoom_WaterLevel rules) =====

// Utility room water heater refill valve (water_heater_fill). ERROR means the ESP8266 either had to
// use its failsafe timer to close the valve, or reset while the valve was open - worth knowing about.
rule "UtilityRoom_WaterFill_State changed to ERROR"
	when
		Item UtilityRoom_WaterFill_State changed to "ERROR"
	then
		val actions = getActions("pushover", "pushover:pushover-account:daa1d3d0d7")
		actions.sendMessage("Boiler fill valve controller in ERROR: " + UtilityRoom_WaterFill_Event.state + " - check the valve is closed","Openhab",null,null,null,null,null,null,"oneplus13")
end

// ===== /etc/openhab/rules/alerts.rules (globals near top) =====

var Timer utilityRoomWaterFillOfflineTimer = null
var boolean utilityRoomWaterFillOfflineAlerted = false

// ===== /etc/openhab/rules/alerts.rules (appended) =====

rule "UtilityRoom_WaterFill_Status changed to offline"
	when
		Item UtilityRoom_WaterFill_Status changed to "offline"
	then
		if (utilityRoomWaterFillOfflineTimer !== null) {
			utilityRoomWaterFillOfflineTimer.cancel
			utilityRoomWaterFillOfflineTimer = null
		}
		utilityRoomWaterFillOfflineTimer = createTimer(now.plusHours(36)) [|
			utilityRoomWaterFillOfflineTimer = null
			if (UtilityRoom_WaterFill_Status.state == "offline") {
				utilityRoomWaterFillOfflineAlerted = true
				val actions = getActions("pushover", "pushover:pushover-account:daa1d3d0d7")
				actions.sendMessage("Boiler fill valve controller went offline","Openhab",null,null,null,null,null,null,"oneplus13")
			}
		]
end

rule "UtilityRoom_WaterFill_Status changed to online"
	when
		Item UtilityRoom_WaterFill_Status changed to "online"
	then
		if (utilityRoomWaterFillOfflineTimer !== null) {
			utilityRoomWaterFillOfflineTimer.cancel
			utilityRoomWaterFillOfflineTimer = null
		}
		if (utilityRoomWaterFillOfflineAlerted) {
			utilityRoomWaterFillOfflineAlerted = false
			val actions = getActions("pushover", "pushover:pushover-account:daa1d3d0d7")
			actions.sendMessage("Boiler fill valve controller back online","Openhab",null,null,null,null,null,null,"oneplus13")
		}
end
